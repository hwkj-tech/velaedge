# SQLite + RocksDB Real Console Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the current cloud-edge MVP genuinely usable by persisting cloud Agent/control data in SQLite, persisting edge runtime rules/state in RocksDB, and wiring the remaining console actions to real backend APIs.

**Architecture:** Cloud remains the source of truth for fleet, models, point mappings, protocol connections, collection tasks, algorithms, releases, Agent suggestions, audit records, and latest runtime status. Edge runtime stores desired/applied rule packages, active version, local device shadow, runtime state, and offline upload queues in RocksDB, then reports metrics/events to cloud when connected. The React console becomes an API-backed management application with real create/edit/publish flows instead of static buttons.

**Tech Stack:** Rust 2021, Axum, Tokio, SQLite via `sqlx`, RocksDB via `rocksdb`, serde JSON contracts, React + TypeScript + Vitest, Vite static build.

Create `crates/cloud-control/src/sqlite.rs` with:

```rust
use anyhow::{Context, Result};
use chrono::Utc;
use edge_core::{EdgeConfigPackage, EdgeRuntimeEvent, EdgeRuntimeMetricsSnapshot};
use sqlx::{sqlite::SqlitePoolOptions, Pool, Row, Sqlite};

use crate::{AuditRecord, EdgeNode, ReleaseRecord};

#[derive(Clone)]
pub struct SqliteCloudStore {
    pool: Pool<Sqlite>,
}

impl SqliteCloudStore {
    pub async fn connect(database_url: &str) -> Result<Self> {
        let pool = SqlitePoolOptions::new()
            .max_connections(5)
            .connect(database_url)
            .await
            .context("failed to connect sqlite cloud store")?;
        sqlx::migrate!("./migrations")
            .run(&pool)
            .await
            .context("failed to run sqlite migrations")?;
        Ok(Self { pool })
    }

    pub async fn upsert_edge_node(&self, node: EdgeNode) -> Result<()> {
        let now = Utc::now().to_rfc3339();
        let capabilities_json = serde_json::to_string(&node.capabilities)?;
        sqlx::query(
            r#"
            INSERT INTO edge_nodes(edge_id, display_name, site, capabilities_json, created_at, updated_at)
            VALUES (?1, ?2, ?3, ?4, ?5, ?5)
            ON CONFLICT(edge_id) DO UPDATE SET
              display_name = excluded.display_name,
              site = excluded.site,
              capabilities_json = excluded.capabilities_json,
              updated_at = excluded.updated_at
            "#,
        )
        .bind(&node.edge_id)
        .bind(&node.display_name)
        .bind(&node.site)
        .bind(capabilities_json)
        .bind(now)
        .execute(&self.pool)
        .await?;
        Ok(())
    }

    pub async fn edge_nodes(&self) -> Result<Vec<EdgeNode>> {
        let rows = sqlx::query(
            "SELECT edge_id, display_name, site, capabilities_json FROM edge_nodes ORDER BY edge_id",
        )
        .fetch_all(&self.pool)
        .await?;

        rows.into_iter()
            .map(|row| {
                Ok(EdgeNode {
                    edge_id: row.try_get("edge_id")?,
                    display_name: row.try_get("display_name")?,
                    site: row.try_get("site")?,
                    capabilities: serde_json::from_str(row.try_get::<String, _>("capabilities_json")?.as_str())?,
                })
            })
            .collect()
    }

    pub async fn upsert_config_package(&self, package: EdgeConfigPackage) -> Result<()> {
        let now = Utc::now().to_rfc3339();
        sqlx::query(
            r#"
            INSERT INTO config_packages(edge_id, version, package_json, created_at)
            VALUES (?1, ?2, ?3, ?4)
            ON CONFLICT(edge_id, version) DO UPDATE SET package_json = excluded.package_json
            "#,
        )
        .bind(&package.edge_id)
        .bind(&package.version)
        .bind(serde_json::to_string(&package)?)
        .bind(now)
        .execute(&self.pool)
        .await?;
        Ok(())
    }

    pub async fn latest_config_package_for_edge(
        &self,
        edge_id: &str,
    ) -> Result<Option<EdgeConfigPackage>> {
        let row = sqlx::query(
            "SELECT package_json FROM config_packages WHERE edge_id = ?1 ORDER BY version DESC LIMIT 1",
        )
        .bind(edge_id)
        .fetch_optional(&self.pool)
        .await?;

        row.map(|row| serde_json::from_str(row.try_get::<String, _>("package_json")?.as_str()))
            .transpose()
            .map_err(Into::into)
    }

    pub async fn upsert_runtime_metrics(
        &self,
        snapshot: EdgeRuntimeMetricsSnapshot,
    ) -> Result<()> {
        let now = Utc::now().to_rfc3339();
        sqlx::query(
            r#"
            INSERT INTO runtime_metrics(edge_id, snapshot_json, updated_at)
            VALUES (?1, ?2, ?3)
            ON CONFLICT(edge_id) DO UPDATE SET
              snapshot_json = excluded.snapshot_json,
              updated_at = excluded.updated_at
            "#,
        )
        .bind(&snapshot.edge_id)
        .bind(serde_json::to_string(&snapshot)?)
        .bind(now)
        .execute(&self.pool)
        .await?;
        Ok(())
    }

    pub async fn runtime_metrics(
        &self,
        edge_id: &str,
    ) -> Result<Option<EdgeRuntimeMetricsSnapshot>> {
        let row = sqlx::query("SELECT snapshot_json FROM runtime_metrics WHERE edge_id = ?1")
            .bind(edge_id)
            .fetch_optional(&self.pool)
            .await?;

        row.map(|row| serde_json::from_str(row.try_get::<String, _>("snapshot_json")?.as_str()))
            .transpose()
            .map_err(Into::into)
    }
}
```

```rust
use axum::{extract::State, Json, Router, routing::{get, post}};
use serde::Serialize;

use crate::AppState;

#[derive(Clone, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct AgentSuggestionResponse {
    pub suggestion_id: String,
    pub title: String,
    pub detail: String,
    pub status: String,
    pub source: String,
}

pub fn routes(state: AppState) -> Router {
    Router::new()
        .route("/api/agent/suggestions", get(list_suggestions))
        .route("/api/agent/suggestions/generate", post(generate_suggestions))
        .with_state(state)
}

async fn list_suggestions(State(state): State<AppState>) -> Json<Vec<AgentSuggestionResponse>> {
    Json(state.list_agent_suggestions().await.unwrap_or_default())
}

async fn generate_suggestions(
```

```rust
use std::path::Path;

use anyhow::{Context, Result};
use edge_core::EdgeConfigPackage;
use rocksdb::{ColumnFamilyDescriptor, Options, DB};

const CF_CONFIGS: &str = "configs";
const CF_SYNC_STATE: &str = "sync_state";
const CF_OFFLINE_QUEUE: &str = "offline_queue";

pub struct RocksEdgeRuntimeStore {
    db: DB,
}

impl RocksEdgeRuntimeStore {
    pub fn open(path: impl AsRef<Path>) -> Result<Self> {
        let mut options = Options::default();
        options.create_if_missing(true);
        options.create_missing_column_families(true);
        let db = DB::open_cf_descriptors(
            &options,
            path,
            vec![
                ColumnFamilyDescriptor::new(CF_CONFIGS, Options::default()),
                ColumnFamilyDescriptor::new(CF_SYNC_STATE, Options::default()),
                ColumnFamilyDescriptor::new(CF_OFFLINE_QUEUE, Options::default()),
            ],
        )
        .context("failed to open rocksdb edge runtime store")?;
        Ok(Self { db })
    }

    pub fn put_desired_config(&self, package: &EdgeConfigPackage) -> Result<()> {
        let cf = self.db.cf_handle(CF_CONFIGS).context("missing configs cf")?;
        let key = format!("desired:{}:{}", package.edge_id, package.version);
        self.db.put_cf(cf, key, serde_json::to_vec(package)?)?;
        Ok(())
    }

    pub fn promote_active_config(&self, edge_id: &str, version: &str) -> Result<()> {
        let cf = self.db.cf_handle(CF_SYNC_STATE).context("missing sync_state cf")?;
        self.db.put_cf(cf, format!("active:{edge_id}"), version.as_bytes())?;
        Ok(())
    }

    pub fn active_config(&self, edge_id: &str) -> Result<Option<EdgeConfigPackage>> {
        let state_cf = self.db.cf_handle(CF_SYNC_STATE).context("missing sync_state cf")?;
        let configs_cf = self.db.cf_handle(CF_CONFIGS).context("missing configs cf")?;
        let Some(version) = self.db.get_cf(state_cf, format!("active:{edge_id}"))? else {
            return Ok(None);
        };
        let version = String::from_utf8(version)?;
        let Some(package) = self
            .db
            .get_cf(configs_cf, format!("desired:{edge_id}:{version}"))?
        else {
            return Ok(None);
        };
        Ok(Some(serde_json::from_slice(&package)?))
    }
}
```

Add FIFO queue methods in the same file:

```rust
impl RocksEdgeRuntimeStore {
    pub fn enqueue_offline_payload(&self, queue: &str, payload: &[u8]) -> Result<()> {
        let cf = self
            .db
            .cf_handle(CF_OFFLINE_QUEUE)
            .context("missing offline_queue cf")?;
        let seq = self.db.latest_sequence_number();
        self.db.put_cf(cf, format!("{queue}:{seq:020}"), payload)?;
        Ok(())
    }

    pub fn drain_offline_payloads(&self, queue: &str, limit: usize) -> Result<Vec<Vec<u8>>> {
        let cf = self
            .db
            .cf_handle(CF_OFFLINE_QUEUE)
            .context("missing offline_queue cf")?;
        let prefix = format!("{queue}:");
        let mut result = Vec::new();
        let mut keys = Vec::new();
        for item in self.db.iterator_cf(cf, rocksdb::IteratorMode::Start) {
            let (key, value) = item?;
            if key.starts_with(prefix.as_bytes()) {
                keys.push(key.to_vec());
                result.push(value.to_vec());
                if result.len() == limit {
                    break;
                }
            }
        }
        for key in keys {
            self.db.delete_cf(cf, key)?;
        }
        Ok(result)
    }
}
```
